========================================================
Employee Performance Metrics — DAX Measures
Paste these into Power BI: Modeling > New Measure
Assumes the table is named "EmployeeData" after import.
========================================================

-- CORE COUNTS --

Total Employees =
COUNTROWS(EmployeeData)

Active Employees =
CALCULATE(COUNTROWS(EmployeeData), EmployeeData[Attrition] = "No")

Attrition Count =
CALCULATE(COUNTROWS(EmployeeData), EmployeeData[Attrition] = "Yes")


-- ATTRITION --

Attrition Rate =
DIVIDE([Attrition Count], [Total Employees], 0)

Attrition Rate (Overtime Only) =
CALCULATE([Attrition Rate], EmployeeData[OverTime] = "Yes")

Attrition Rate (No Overtime) =
CALCULATE([Attrition Rate], EmployeeData[OverTime] = "No")

Overtime Attrition Gap =
[Attrition Rate (Overtime Only)] - [Attrition Rate (No Overtime)]


-- PERFORMANCE --

Avg Performance Rating =
AVERAGE(EmployeeData[PerformanceRating])

High Performers =
CALCULATE(COUNTROWS(EmployeeData), EmployeeData[PerformanceRating] >= 3)

High Performer Rate =
DIVIDE([High Performers], [Total Employees], 0)

Avg Job Satisfaction =
AVERAGE(EmployeeData[JobSatisfaction])

Avg Work-Life Balance =
AVERAGE(EmployeeData[WorkLifeBalance])


-- COMPENSATION --

Avg Monthly Income =
AVERAGE(EmployeeData[MonthlyIncome])

Median Monthly Income =
MEDIAN(EmployeeData[MonthlyIncome])

Income Percentile Rank (per employee, use as calculated column instead) =
RANKX(ALL(EmployeeData), EmployeeData[MonthlyIncome], , ASC, DENSE) 
    -- add as a calculated column, not a measure, if you need per-row rank


-- TENURE & GROWTH --

Avg Years At Company =
AVERAGE(EmployeeData[YearsAtCompany])

Avg Years Since Promotion =
AVERAGE(EmployeeData[YearsSinceLastPromotion])

Overdue for Promotion Count =
CALCULATE(COUNTROWS(EmployeeData), EmployeeData[YearsSinceLastPromotion] >= 4)


-- FLIGHT RISK SCORE (calculated column, add via Modeling > New Column) --

FlightRiskScore =
VAR SatPenalty = (5 - EmployeeData[JobSatisfaction]) * 2
VAR WLBPenalty = (5 - EmployeeData[WorkLifeBalance]) * 2
VAR OTPenalty = IF(EmployeeData[OverTime] = "Yes", 3, 0)
VAR TravelPenalty = IF(EmployeeData[BusinessTravel] = "Travel_Frequently", 2, 0)
VAR TenureBonus = MIN(EmployeeData[YearsAtCompany], 10) * -0.3
RETURN
    SatPenalty + WLBPenalty + OTPenalty + TravelPenalty + TenureBonus

-- Then bucket it with a calculated column:
FlightRiskTier =
SWITCH(
    TRUE(),
    EmployeeData[FlightRiskScore] >= 12, "High Risk",
    EmployeeData[FlightRiskScore] >= 7,  "Medium Risk",
    "Low Risk"
)

High Flight Risk Count =
CALCULATE(COUNTROWS(EmployeeData), EmployeeData[FlightRiskTier] = "High Risk")


-- DEPARTMENT COMPARISON --

Dept Avg Performance =
CALCULATE([Avg Performance Rating], ALLEXCEPT(EmployeeData, EmployeeData[Department]))

Performance vs Dept Avg =
[Avg Performance Rating] - [Dept Avg Performance]
